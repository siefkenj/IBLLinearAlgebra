% transforming: the image of the unit circle is that same circle under a

% eigenvectors off the 45 degree directions simply lists their angles.

% The pretext conversion lifts that block out and hands it to pretext as

%%%
% Palette.  A vector's hue is a function of its starting angle, sampled
% from a cyclic ramp, so an eigenvector inserted between two of the
% standard directions lands between them in colour as well.
%
% The ramp is an ellipse in the (a,b) plane of CIE Lab with a sinusoidal
% lightness, fitted so that the eight sampled hues stay distinguishable
% under normal vision and under simulated protanopia, deuteranopia and
% tritanopia: the smallest pairwise Lab distance over all four is 16.9.
% Lightness is capped so every hue stays legible as a line on white.
%%%
\definecolor{EVdirA}{HTML}{7ABD7D}  %   0 degrees
\definecolor{EVdirB}{HTML}{10ACA0}  %  45
\definecolor{EVdirC}{HTML}{00769A}  %  90
\definecolor{EVdirD}{HTML}{2A3D71}  % 135
\definecolor{EVdirE}{HTML}{531443}  % 180
\definecolor{EVdirF}{HTML}{761E28}  % 225
\definecolor{EVdirG}{HTML}{995C25}  % 270
\definecolor{EVdirH}{HTML}{A19D47}  % 315
% eigendirections that fall between two standard ones, same ramp
\definecolor{EVeigA}{HTML}{47B897}  %  28.155
\definecolor{EVeigB}{HTML}{005183}  % 118.155
\definecolor{EVeigC}{HTML}{43B798}  %  30

% A starting vector is dashed and its image solid.  On top of that, a
% vector is marked by which side of the y-axis it points to: one dot if
% it points right, a tight pair of dots if it points left, nothing if it
% lies along the y-axis.  A starting vector carries its marks after two
% short dashes, its image after one long dash.
%
% Taking the long dash to be the two short dashes with the gap between
% them filled in is what makes the marks line up.  With short dash a, gap
% b, dot d and long dash L = 2a + b, the marks begin at 2a + 2b on both
% patterns and the two patterns share a period.  So the dots fall the
% same distance from the origin along a vector and along its image, which
% is what lets the two be matched by eye when they are nearly parallel.
%
% The gap b is as narrow as it can be: below about 2pt the two short
% dashes read as one long one and a marked starting vector stops being
% distinguishable from its image.  The gap g inside a double dot has to
% stay well under b, or the pair reads as two separate dots.
%
% These are lengths on the page, not diagram coordinates, so a figure
% drawn at a smaller \EVunit rescales them to keep the same number of
% dashes along a vector.  \EVpanelsizes does this.
%
% The starting vectors are drawn at \EVoriginopacity so that where an
% image crosses one it is the image that reads as the solid mark.  The
% images are fully opaque.
%
% EVeigendash is the older heavy dash for an eigenvector's starting
% vector, kept for \EVemphasiseeigentrue.
%
% Every arrow in these figures is drawn at \EVvecwidth.  It is a width on
% the page and is deliberately not scaled down for the smaller panels, so
% a hairline never appears.
\newcommand{\EVvecwidth}{1.5pt}
\newcommand{\EVoriginopacity}{0.6}
\newcommand{\EVdasha}{4.75pt} % short dash
\newcommand{\EVdashb}{1.5pt}  % gap between marks
\newcommand{\EVdashd}{1.5pt}  % a dot
\newcommand{\EVdashg}{0.8pt}  % gap inside a double dot
\newcommand{\EVdashp}{3pt}    % dash and gap of an unmarked vector
% The long dash is not a free parameter: it is the two short dashes with
% the gap between them filled in.  Deriving it here rather than writing a
% number down means the marks cannot drift out of line when a, b or the
% panel scale is changed.
\newcommand{\EVdashL}{\the\dimexpr\EVdasha*2+\EVdashb\relax}
\tikzset{
	EVplaindash/.style={dash pattern=on \EVdashp off \EVdashp,
		line width=\EVvecwidth, opacity=\EVoriginopacity},
	EVsolid/.style={line width=\EVvecwidth},
	EVrightorig/.style={dash pattern=on \EVdasha off \EVdashb on \EVdasha
		off \EVdashb on \EVdashd off \EVdashb, line width=\EVvecwidth,
		opacity=\EVoriginopacity},
	EVrightimg/.style={dash pattern=on \EVdashL off \EVdashb
		on \EVdashd off \EVdashb, line width=\EVvecwidth},
	EVleftorig/.style={dash pattern=on \EVdasha off \EVdashb on \EVdasha
		off \EVdashb on \EVdashd off \EVdashg on \EVdashd off \EVdashb,
		line width=\EVvecwidth, opacity=\EVoriginopacity},
	EVleftimg/.style={dash pattern=on \EVdashL off \EVdashb
		on \EVdashd off \EVdashg on \EVdashd off \EVdashb,
		line width=\EVvecwidth},
	EVeigendash/.style={dash pattern=on 5pt off 3pt, line width=1.8pt,
		opacity=\EVoriginopacity},
}

% How far apart a colliding pair ends up.  The two are separated
% symmetrically about the line they share: the starting vector steps a
% quarter turn negatively, its image a quarter turn positively, each by
% half the gap, so neither sits where the true vector was.

%%%
% The grey frame shared by the square picture and the circle picture, so
% their gridlines agree.  The grid is drawn on integer coordinates well
% past the reach and then clipped, which both keeps the spacing at 1 and
% lets it run off every edge of the figure.
%%%
\newcommand{\EVframe}{%
	\clip (-\EVreach,-\EVreach) rectangle (\EVreach,\EVreach);
	\draw[EVaxiscol,thin] (-6,-6) grid[step=1] (6,6);
	\draw[EVaxiscol,semithick]
		(-\EVreach,0) -- (\EVreach,0)  (0,-\EVreach) -- (0,\EVreach);
}

%%%
% Before/after halves of a circle diagram, for the side-by-side figure.
% \EVcircleBefore{extra eigendirections}{standard ones that are eigen}
% \EVcircleAfter[directions]{a}{b}{c}{d}{extra eigendirections}
%
% The after half takes its own direction table so that a transformation
% which sends some direction to zero can leave that one out; the before
% half always draws all eight, since every starting vector is a vector.
%
% Both halves are only ever drawn as panels of \EVTransformPair, so they
% take the panel sizes directly.
%%%
\newcommand{\EVcircleBefore}[2]{%
	\begin{tikzpicture}[x=\EVpanelunit,y=\EVpanelunit]
		\EVpanelsizes
		\EVframe
		\draw[EVcirclecol,dashed,thick] (0,0) circle[radius=1];
		\foreach \ang/\col/\os/\is in \EVdirections {%
			\draw[\col,\os,->] (0,0) -- (\ang:1);
		}
		\foreach \i/\ang/\col/\os/\is/\shift in {#1}{%
			\draw[\col,\os,->] (0,0) -- (\ang:1);
		}
		\ifEVemphasiseeigen
			\foreach \ang/\col in {#2}{%
				\draw[\col,EVeigendash,->] (0,0) -- (\ang:1);
			}
		\fi
	\end{tikzpicture}%
}
\NewDocumentCommand{\EVcircleAfter}{O{\EVdirections} m m m m m}{%
	\begin{tikzpicture}[x=\EVpanelunit,y=\EVpanelunit]
		\EVpanelsizes
		\EVframe
		\draw[EVcirclecol,thick,cm={#2,#4,#3,#5,(0,0)}] (0,0) circle[radius=1];
		\begin{scope}[cm={#2,#4,#3,#5,(0,0)}]
			\foreach \ang/\col/\os/\is in #1 {\path (\ang:1) coordinate (v\ang);}
			\foreach \i/\ang/\col/\os/\is/\shift in {#6} {\path (\ang:1) coordinate (e\i);}
		\end{scope}
		\foreach \ang/\col/\os/\is in #1 {\draw[\col,\is,->] (0,0) -- (v\ang);}
		\foreach \i/\ang/\col/\os/\is/\shift in {#6}{\draw[\col,\is,->] (0,0) -- (e\i);}
	\end{tikzpicture}%
}

%%%
% \EVcircle[starts][images]{a}{b}{c}{d}{extra eigendirections}{standard
%                                                     ones that are eigen}
% #1 and #2 are the angle/colour/shift/dash tables above: which starting
% vectors are landed on, and which images land on something.  #7 is a
% possibly empty list of index/angle/colour/dash/dash/shift entries for
% eigenvectors off the eight standard directions -- one shift serves for
% both halves of the pair.  #8 is used only under \EVemphasiseeigentrue.
%
% Every arrow is drawn from its own start outwards, so a marked pair puts
% its dots the same distance along both halves however far the pair has
% been stepped apart.
%%%
%%%
% \EVfit{<material>}
%
% Typeset <material> as one box, but start a new page first if the whole of it
% will not fit in what is left of this one.
%
% Without this, TeX will squeeze a picture onto the bottom of a page by using
% up the page's available shrink, which leaves the bottom of the picture
% sitting under the footer.  Shrinking a page is legal, so this happens
% silently: there is no `Overfull \vbox' to warn about it, even at \vfuzz=0pt.
%
% \pagegoal is \maxdimen until something has been contributed to the page.
% Nothing ever is in a standalone latex-image compile, so the test is skipped
% there and the macro stays a plain box, which is what the pretext build needs.
%%%
\newsavebox{\EVfitbox}
\newcommand{\EVfit}[1]{%
	\sbox\EVfitbox{#1}%
	\ifdim\pagegoal<\maxdimen
		\ifdim\dimexpr\pagetotal+\ht\EVfitbox+\dp\EVfitbox+2\topsep\relax>\pagegoal
			\newpage
		\fi
	\fi
	\usebox\EVfitbox
}

\NewDocumentCommand{\EVcircle}{O{\EVstartsNoneOnTop} O{\EVimagesNoneOnTop} m m m m m m}{%
	\EVfit{%
	\begin{tikzpicture}[x=\EVunit,y=\EVunit]
		\EVframe
		% the unit circle and its image, both grey
		\draw[EVcirclecol,dashed,thick] (0,0) circle[radius=1];
		\draw[EVcirclecol,thick,cm={#3,#5,#4,#6,(0,0)}] (0,0) circle[radius=1];
		% Read the images off inside the transformation, then draw the
		% arrows outside it so that the heads are not sheared.
		\begin{scope}[cm={#3,#5,#4,#6,(0,0)}]
			\foreach \ang/\col/\os/\is in \EVdirections {\path (\ang:1) coordinate (v\ang);}
			\foreach \i/\ang/\col/\os/\is/\shift in {#7} {\path (\ang:1) coordinate (e\i);}
		\end{scope}
		% Starting vectors, stepped a quarter turn negatively when an
		% image is going to land on them.
		\foreach \ang/\col/\shift/\os in #1 {%
			\path ($(0,0)!\shift!-90:(\ang:1)$) coordinate (s\ang);
			\draw[\col,\os,->] (s\ang) -- ($(\ang:1)+(s\ang)$);
		}
		% Images, stepped the same distance the other way, about the
		% direction they landed in rather than the one they came from.
		\foreach \ang/\col/\shift/\is in #2 {%
			\path ($(0,0)!\shift!90:(v\ang)$) coordinate (o\ang);
			\draw[\col,\is,->] (o\ang) -- ($(v\ang)+(o\ang)$);
		}
		% Eigendirections that are not among the eight: both halves here.
		\foreach \i/\ang/\col/\os/\is/\shift in {#7}{%
			\path ($(0,0)!\shift!-90:(\ang:1)$) coordinate (g\i);
			\ifEVemphasiseeigen
				\draw[\col,EVeigendash,->] (g\i) -- ($(\ang:1)+(g\i)$);
			\else
				\draw[\col,\os,->] (g\i) -- ($(\ang:1)+(g\i)$);
			\fi
			\path ($(0,0)!\shift!90:(e\i)$) coordinate (f\i);
			\draw[\col,\is,->] (f\i) -- ($(e\i)+(f\i)$);
		}
		\ifEVemphasiseeigen
			\foreach \ang/\col in {#8}{%
				\draw[\col,EVeigendash,->] (0,0) -- (\ang:1);
			}
		\fi
	\end{tikzpicture}%
	}%
}

%%%
% \EVgrid{a}{b}{c}{d}
% The unit square under A, drawn on the same frame as the circle
% pictures.  Straight lines stay straight, so this is a plain tikz grid
% drawn inside a cm transformation.  The square carries no internal
% gridlines of its own -- the grey grid behind it supplies the spacing --
% only its outline, its quadrant tints, and the images of the two axes.
%
% Like the circle halves, this is only ever drawn as a panel.
%%%
\newcommand{\EVgrid}[4]{%
	\begin{tikzpicture}[x=\EVpanelunit,y=\EVpanelunit]
		\EVpanelsizes
		\EVframe
		\begin{scope}[cm={#1,#3,#2,#4,(0,0)}]
			% each quadrant takes the ramp hue of its own bisector, so the
			% square follows the same angular colour flow as the vectors
			\foreach \sx/\sy/\col in {1/1/EVdirB, -1/1/EVdirD,
				-1/-1/EVdirF, 1/-1/EVdirH}{%
				\fill[\col,opacity=0.16] (0,0) rectangle (\sx,\sy);
				\draw[\col,thin] (0,0) rectangle (\sx,\sy);
			}
			% images of the two axes
			\draw[very thick] (-1,0) -- (1,0);
			\draw[very thick,dashed] (0,-1) -- (0,1);
		\end{scope}
	\end{tikzpicture}%
}

%%%
% The book annotates "this maps to that" with a gently bent arrow
% carrying the transformation above it (see module8 and module14).
%%%
\newcommand{\EVmapsto}[1]{%
	\begin{tikzpicture}
		\draw[thick] (0,0) edge[bend left, ->] node[midway,above] {#1} (0.9,0);
	\end{tikzpicture}%
}

%%%
% \EVTransformPair{a}{b}{c}{d}
% The square and the unit circle, each shown beside its image under A.
% Four panels and two arrows come to about 420pt against a 433pt text
% width, so the two transforms sit on one line.  \vcenter puts every
% cell on the math axis so the panels and the arrows line up.
%
% The panels are drawn at 22pt to the unit against the 42pt of the
% overlay figures.  Each panel picture installs \EVpanelsizes itself, so
% there is nothing to set up or undo here.
%
% The optional argument is the direction table for the image panel, for
% the case where A sends one of the eight to zero.
%
% The macro expands to a single box and does no centring of its own, so
% that it is legal where only a box is: the pretext build compiles each
% figure on its own in a standalone document, and `center' is not allowed
% there.  The call sites supply the surrounding `center'.
%%%
\NewDocumentCommand{\EVTransformPair}{O{\EVdirections} m m m m}{%
	\EVfit{%
	\begin{tabular}{c@{\hspace{8pt}}c@{\hspace{8pt}}c@{\hspace{20pt}}c@{\hspace{8pt}}c@{\hspace{8pt}}c}
		\multicolumn{3}{c}{\small Transforming a Square}
		& \multicolumn{3}{c}{\small Transforming the Unit Circle}
		\\[4pt]
		\(\vcenter{\hbox{\EVgrid{1}{0}{0}{1}}}\)
		& \(\vcenter{\hbox{\EVmapsto{\(\mathcal{T}_A\)}}}\)
		& \(\vcenter{\hbox{\EVgrid{#2}{#3}{#4}{#5}}}\)
		& \(\vcenter{\hbox{\EVcircleBefore{}{}}}\)
		& \(\vcenter{\hbox{\EVmapsto{\(\mathcal{T}_A\)}}}\)
		& \(\vcenter{\hbox{\EVcircleAfter[#1]{#2}{#3}{#4}{#5}{}}}\)
	\end{tabular}%
	}%
}

\Heading{Visualizing Eigenvectors}

Real eigenvectors for a transformation $T$ can be seen by comparing vectors in the unit circle, $\mathcal{S}$, and vectors in its image, $T(\mathcal{S})$,
 and looking for those vectors that don't change direction.

\subsubsection*{Uniform Scaling}
Consider the dilation $\mathcal{T}_{A}$ where $A=\matc{3/2 & 0\\ 0 & 3/2}$. This transformation scales all vectors by a factor of $3/2$.
We can visualize this transformation by studying how a region is transformed\footnote{ All we really need is to look at two linear independent vectors, but looking at more vectors doesn't hurt!}. In this case, we will focus on a square region and the unit 
circle.

\begin{center}
	\EVTransformPair{1.5}{0}{0}{1.5}
\end{center}

Overlaying the unit circle with its image, we can see that every non-zero vector's direction is preserved, and so all non-zero vectors are eigenvectors. 


\begin{center}
	\EVcircle[\EVstartsAllOnTop][\EVimagesAllOnTop]{1.5}{0}{0}{1.5}{}{0/EVdirA, 45/EVdirB, 90/EVdirC, 135/EVdirD, 180/EVdirE, 225/EVdirF, 270/EVdirG, 315/EVdirH}
\end{center}

In the diagrams, the following conventions are used: vectors on the original unit circle are dashed (with occasional dots) 
while corresponding vectors on the image of the unit circle are solid (with occasional dots); vectors are colored based on their angle;
and dot patterns are provided to help you distinguish between vectors with similar colors. Additionally, vectors that normally would be drawn directly
on top of one another have been moved slightly so that their tails don't overlap.

\subsubsection*{Stretch along an Axis}
Consider $\mathcal{T}_{A}$, where $A=\mat{3/2 & 0\\ 0 & 1}$. This transformation stretches vectors by a factor of $3/2$ in the
$\xhat$-direction and leaves the $\yhat$-direction alone.

\begin{center}
	\EVTransformPair{1.5}{0}{0}{1}
\end{center}

Overlaying the unit circle with its image, we can see that only the vectors lying along the coordinate axes keep their direction.
Non-zero multiples of $\xhat$ and $\yhat$ are eigenvectors (with eigenvalues $3/2$ and $1$, respectively), while all other vectors are not eigenvectors.

\begin{center}
	\EVcircle[\EVstartsAxisDilation][\EVimagesAxisDilation]{1.5}{0}{0}{1}{}{0/EVdirA, 90/EVdirC, 180/EVdirE, 270/EVdirG}
\end{center}

\subsubsection*{Reflection}
Consider the reflection $\mathcal{T}_{A}$, where $A=\mat{0 & 1\\ 1 & 0}$. This transformation reflects every vector across the
line with direction $\xhat + \yhat$.

\begin{center}
	\EVTransformPair{0}{1}{1}{0}
\end{center}

Overlaying the unit circle with its image, we see that every vector in the unit circle is sent to another vector in the unit circle; however, studying closely,
only vectors pointing along the lines with directions $\xhat+\yhat$ and $\xhat - \yhat$ keep their direction.
All other vectors change direction.

\begin{center}
	\EVcircle[\EVstartsAllOnTop][\EVimagesAllOnTop]{0}{1}{1}{0}{}{45/EVdirB, 135/EVdirD, 225/EVdirF, 315/EVdirH}
\end{center}

\subsubsection*{Shear}
Consider the shear $\mathcal{T}_{A}$, where $A=\mat{1 & 1/2\\ 0 & 1}$. This transformation slides every vector sideways by 
$1/2$ its $x$ coordinate.

\begin{center}
	\EVTransformPair{1}{0.5}{0}{1}
\end{center}

Overlaying the unit circle with its image, we can see that only the vectors lying along the $x$-axis keep their direction (they don't move at all),
while all other vectors change direction.

\begin{center}
	\EVcircle[\EVstartsShear][\EVimagesShear]{1}{0.5}{0}{1}{}{0/EVdirA, 180/EVdirE}
\end{center}

\subsubsection*{Projection}
Consider the projection $\mathcal{T}_{A}$, where $A=\mat{1/2 & 1/2\\ 1/2 & 1/2}$. This transformation projects every vector onto
the line $y=x$, flattening the whole plane onto that single line.

\begin{center}
	\EVTransformPair[\EVdirectionsProjection]{0.5}{0.5}{0.5}{0.5}
\end{center}

Overlaying the unit circle with its image, we can see that non-zero vectors lying along the line with direction $\xhat + \yhat$ remain unchanged,
and so are eigenvectors
with eigenvalue $1$. Non-zero vectors along the line with direction $\xhat - \yhat$ are all sent to $\vec 0$, and so are eigenvectors with
eigenvalue $0$. All other non-zero vectors change direction when transformed, and so are not eigenvectors.

\begin{center}
	\EVcircle[\EVstartsProjection][\EVimagesProjection]{0.5}{0.5}{0.5}{0.5}{}{45/EVdirB, 225/EVdirF}
\end{center}

\subsubsection*{Rotation}
Consider the rotation $\mathcal{T}_{A}$, where $A=\mat{\cos 10^{\circ} & -\sin 10^{\circ}\\ \sin 10^{\circ} & \cos 10^{\circ}}$. This transformation rotates every
vector counter-clockwise by $10^{\circ}$.

\begin{center}
	\EVTransformPair{0.984808}{-0.173648}{0.173648}{0.984808}
\end{center}

Overlaying the unit circle with its image, we can see that the unit circle maps to itself. However, since every non-zero vector was rotated, this
transformation has no \emph{real} eigenvectors.

\begin{center}
	\EVcircle[\EVstartsNoneOnTop][\EVimagesNoneOnTop]{0.984808}{-0.173648}{0.173648}{0.984808}{}{}
\end{center}
